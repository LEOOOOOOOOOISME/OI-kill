## 二、引擎规则改动（src/engine/core.js，函数+行号）

### 规则 1：内奸单挑回血「每回合回 1」→「每 2 回合回 1 点」

- 玩家状态初始化（`setup`，行 244）：新增 `duelTurns: 0`（内奸单挑回合计数）。
- `startTurn`（行 336 起，规则块行 358–367）：

```js
// 旧(H12):
// 内奸单挑: 存活2人且含内奸 -> 每回合回1(H12)
if (p.identity === 'traitor') {
  const alive = g.players.filter(q => !q.dead);
  if (alive.length === 2) {
    p.hp = Math.min(p.maxHp, p.hp + 1);
    g.log.push({ t: g.round, txt: `${p.name}【单挑】每回合开始回复1体力`, cls: 'act' });
  }
}

// 新(Fix-Balance R1):
// 内奸单挑: 存活2人且含内奸 -> 每2回合回1点(Fix-Balance R1, 用户裁决A; 原H12为每回合回1)
if (p.identity === 'traitor') {
  const alive = g.players.filter(q => !q.dead);
  if (alive.length === 2) {
    p.duelTurns++;
    if (p.duelTurns % 2 === 0) {
      p.hp = Math.min(p.maxHp, p.hp + 1);
      g.log.push({ t: g.round, txt: `${p.name}【单挑】第${p.duelTurns}回合开始回复1体力(每2回合1点)`, cls: 'act' });
    }
  }
}
```

- 触发条件保持原样（存活 2 人且含内奸，即内奸 vs 主公单挑阶段），仅把回血频率降为单挑开始后第 2、4、6… 个自己的回合。无牌张影响。

### 规则 2：保底终局平票「内奸单独胜」→「再战一轮，仍平票判主公方」

- 对局状态初始化（`createGame`，行 88）：新增 `tiebreakExtra: false`（再战一轮标记）。
- `forceEndByCount`（行 654–683）重写平票分支：

```js
// 旧:
} else if (traitorCnt > 0) {
  // 人数相同且内奸存活 -> 内奸单独获胜(FAQ#15)
  g.log.push({ t: g.round, txt: `【保底终局】人数相同,内奸单独获胜!`, cls: 'evt' });

} else {
  // 无内奸平票: 记平局按主公方(兜底)

  end(g, 'lord');
}

// 新:
if (winners.length === 1) { ...end(g, winners[0]); return true; }        // 人数多者胜, 不变
if (!g.tiebreakExtra) {
  // 平票(含内奸存活): 再战一轮(用户裁决A), 对局继续
```

- 函数现返回「对局是否结束」：`true`=已判胜负，`false`=再战一轮。三处调用点（行 292 `draw` / 行 509、526 `endTurn` 轮转）由

  `if (g.reshuffleCount >= 2 && forceEndByCount(g)) return;` —— **再战一轮时不再提前 return，摸牌/轮转照常继续**，对局无卡顿、无牌张丢失。
- 「人数多者胜」分支不变；无内奸平票同样先再战一轮（覆盖 DECISIONS #8 的旧兜底）。所有终局路径仍走 `end()`（行 739 起），Fix-P10 的清扫（sweepTransientCards + 清空 prompts/_promptGroups）依旧生效，终局守恒 120 与提示残留清零不受影响。
